Integração Google Sheets

Este aplicativo utiliza o Google Sheets como banco de dados principal. Para conectar a sua planilha existente:

  1. Abra sua planilha no Google Drive.
  2. Acesse Extensões > Apps Script.
  3. Cole o código Apps Script fornecido abaixo e clique em Implantar > Nova Implantação (Web App).
  4. Escolha permissão: "Qualquer pessoa" (Anyone).
  5. Copie o URL da Web App gerado e cole no campo abaixo.
Código Apps Script para a Planilha: 
function doGet(e) {
  return handleRequest(e);
}
function doPost(e) {
  return handleRequest(e);
}
function handleRequest(e) {
  var lock = LockService.getScriptLock();
  lock.tryLock(10000);
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    var param = e.parameter.action || (e.postData ? JSON.parse(e.postData.contents).action : '');
    
    if (param === 'readAll') {
      var data = {
        cards: getSheetData(ss, 'Cartões'),
        responsibles: getSheetData(ss, 'Responsáveis'),
        categories: getSheetData(ss, 'Categorias'),
        expenses: getSheetData(ss, 'Despesas'),
        installments: getSheetData(ss, 'Parcelas')
      };
      return responseJSON(data);
    }
    
    if (e.postData) {
      var payload = JSON.parse(e.postData.contents);
      if (payload.action === 'syncAll') {
        saveSheetData(ss, 'Cartões', payload.cards);
        saveSheetData(ss, 'Responsáveis', payload.responsibles);
        saveSheetData(ss, 'Categorias', payload.categories);
        saveSheetData(ss, 'Despesas', payload.expenses);
        saveSheetData(ss, 'Parcelas', payload.installments);
        return responseJSON({ status: 'success' });
      }
    }
    return responseJSON({ status: 'ok' });
  } catch(err) {
    return responseJSON({ error: err.toString() });
  } finally {
    lock.releaseLock();
  }
}
function getSheetData(ss, sheetName) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) return [];
  var values = sheet.getDataRange().getValues();
  if (values.length <= 1) return [];
  var headers = values[0];
  var result = [];
  for (var i = 1; i < values.length; i++) {
    var row = {};
    for (var j = 0; j < headers.length; j++) {
      row[headers[j]] = values[i][j];
    }
    result.push(row);
  }
  return result;
}
function saveSheetData(ss, sheetName, dataArray) {
  var sheet = ss.getSheetByName(sheetName);
  if (!sheet) sheet = ss.insertSheet(sheetName);
  sheet.clearContents();
  if (!dataArray || dataArray.length === 0) return;
  var headers = Object.keys(dataArray[0]);
  sheet.appendRow(headers);
  var rows = dataArray.map(function(item) {
    return headers.map(function(h) { return item[h] !== undefined ? item[h] : ''; });
  });
  sheet.getRange(2, 1, rows.length, headers.length).setValues(rows);
}
function responseJSON(obj) {
  return ContentService.createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
